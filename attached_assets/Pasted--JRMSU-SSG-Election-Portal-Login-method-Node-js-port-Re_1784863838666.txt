// JRMSU SSG Election Portal — Login method (Node.js port)
// Requires: node >= 18 (native fetch) or install node-fetch
// Copy .env.example to .env and fill in your values before running.
//
// Usage:
//   node index.js                          → runs the built-in demo
//   const { login } = require('./index')   → import into your own project

require('dotenv').config();

// ── Constants ─────────────────────────────────────────────────────────────────
const ARMS_TOKEN_URL = 'https://jrmsu-arms.online/api/version-2/services/credential/token/request';
const ARMS_LOGIN_URL = 'https://jrmsu-arms.online/api/version-2/services/student/account/login';

const BASE_HEADERS = {
  'User-Agent': 'Coderstation-Protocol',
  'Referer':    'https://jrmsu-election-system.vercel.app/',
  'Origin':     'https://jrmsu-election-system.vercel.app',
};

// ── Step 1: Request an ARMS bearer token ─────────────────────────────────────
async function armsGetToken() {
  const apiKey    = process.env.ARMS_API_KEY;
  const apiSecret = process.env.ARMS_API_SECRET;

  if (!apiKey || !apiSecret) {
    throw new Error('Missing ARMS_API_KEY or ARMS_API_SECRET in environment.');
  }

  const res = await fetch(ARMS_TOKEN_URL, {
    method:  'POST',
    headers: {
      ...BASE_HEADERS,
      'Api-Key':    apiKey,
      'Api-Secret': apiSecret,
    },
  });

  const data = await res.json();

  if (!res.ok) {
    throw new Error(`Token request failed (${res.status}): ${data?.message ?? 'Unknown error'}`);
  }

  const secretKey = data.Secret_Key ?? data.SecretKey ?? data.secretKey ?? null;
  const jwToken   = data.JWToken    ?? data.Token     ?? data.jwToken   ?? null;

  if (!secretKey || !jwToken) {
    throw new Error('ARMS token response did not include Secret_Key / JWToken.');
  }

  return { secretKey, jwToken };
}

// ── Step 2: Authenticate the student ─────────────────────────────────────────
async function armsLogin(secretKey, jwToken, studentId, password) {
  const res = await fetch(ARMS_LOGIN_URL, {
    method:  'POST',
    headers: {
      ...BASE_HEADERS,
      'Secret-Key':    secretKey,
      'Token':         jwToken,
      'Authorization': `Bearer ${jwToken}`,
      'Content-Type':  'application/json',
    },
    body: JSON.stringify({ Username: studentId, Password: password }),
  });

  const data = await res.json();

  if (!res.ok || !data?.Record) {
    throw new Error(`Login failed (${res.status}): ${data?.message ?? 'Incorrect Student ID or password.'}`);
  }

  return data.Record;
}

// ── Enrollment check (mirrors PHP logic) ─────────────────────────────────────
function checkEnrollment(record) {
  const raw = record.Enrollment_Status
           ?? record.enrollment_status
           ?? record.EnrollmentStatus
           ?? null;

  if (raw === null) return true; // field absent → allow (same as PHP)

  const status = String(raw).toLowerCase().trim();
  return status.includes('enroll') || status === 'active';
}

// ── Main export: login(studentId, password) ───────────────────────────────────
/**
 * Authenticate a JRMSU student via the ARMS API.
 *
 * @param {string} studentId  e.g. "25-A-00000"
 * @param {string} password
 * @returns {Promise<object>} Student record from ARMS on success.
 * @throws  {Error}           Descriptive error on any failure.
 */
async function login(studentId, password) {
  if (!studentId || !password) {
    throw new Error('Student ID and password are required.');
  }

  // Step 1 — get token
  const { secretKey, jwToken } = await armsGetToken();

  // Step 2 — student login
  const record = await armsLogin(secretKey, jwToken, studentId, password);

  // Step 3 — enrollment gate
  if (!checkEnrollment(record)) {
    const status = record.Enrollment_Status ?? record.enrollment_status ?? 'Unknown';
    throw new Error(`Access denied. Enrollment status from ARMS: "${status}". Only enrolled students may vote.`);
  }

  // Return only the fields relevant to the election portal session
  return {
    studentId:        record.Student_ID       ?? record.student_id   ?? studentId,
    studentName:      record.Student_Name     ?? record.student_name ?? '',
    sex:              record.Sex              ?? record.Gender        ?? '',
    email:            record.Email            ?? '',
    mobile:           record.Mobile           ?? '',
    college:          record.College          ?? '',
    collegeCode:      record.College_Code     ?? record.college_code ?? '',
    programEnrolled:  record.Program_Enrolled ?? record.program      ?? '',
    programCode:      record.Program_Code     ?? record.program_code ?? '',
    yearLevel:        record.Year_Level       ?? record.year_level   ?? '',
    semester:         record.Semester         ?? '',
    schoolYear:       record.School_Year      ?? '',
    enrollmentStatus: record.Enrollment_Status ?? record.enrollment_status ?? '',
  };
}

// ── Quick demo (runs only when called directly: node index.js) ────────────────
if (require.main === module) {
  const studentId = process.env.TEST_STUDENT_ID;
  const password  = process.env.TEST_PASSWORD;

  if (!studentId || !password) {
    console.error('Set TEST_STUDENT_ID and TEST_PASSWORD in your .env to run the demo.');
    process.exit(1);
  }

  console.log(`Logging in as: ${studentId} …`);
  login(studentId, password)
    .then(record => {
      console.log('\n✅ Login successful!\n');
      console.log(JSON.stringify(record, null, 2));
    })
    .catch(err => {
      console.error('\n❌ Login failed:', err.message);
      process.exit(1);
    });
}

module.exports = { login };
